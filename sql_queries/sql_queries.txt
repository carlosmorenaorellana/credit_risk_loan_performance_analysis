# 4 -- Validate data types using SAFE_CAST and regex checks to detect formatting issues or invalid characters
SELECT 
  -- Integer validations
  COUNTIF(SAFE_CAST(person_age AS INT64) IS NULL AND person_age IS NOT NULL) AS invalid_person_age,
  COUNTIF(SAFE_CAST(person_income AS INT64) IS NULL AND person_income IS NOT NULL) AS invalid_person_income,
  COUNTIF(SAFE_CAST(loan_amnt AS INT64) IS NULL AND loan_amnt IS NOT NULL) AS invalid_loan_amnt,
  COUNTIF(SAFE_CAST(loan_status AS INT64) IS NULL AND loan_status IS NOT NULL) AS invalid_loan_status,
  COUNTIF(SAFE_CAST(cb_person_cred_hist_length AS INT64) IS NULL AND cb_person_cred_hist_length IS NOT NULL) AS invalid_cred_hist_length,

  -- Float validations
  COUNTIF(SAFE_CAST(person_emp_length AS FLOAT64) IS NULL AND person_emp_length IS NOT NULL) AS invalid_person_emp_length,
  COUNTIF(SAFE_CAST(loan_int_rate AS FLOAT64) IS NULL AND loan_int_rate IS NOT NULL) AS invalid_loan_int_rate,
  COUNTIF(SAFE_CAST(loan_percent_income AS FLOAT64) IS NULL AND loan_percent_income IS NOT NULL) AS invalid_loan_percent_income,

  -- String / Boolean validations (checking for unexpected numeric digits or formatting issues)
  COUNTIF(REGEXP_CONTAINS(CAST(person_home_ownership AS STRING), r'[0-9]')) AS invalid_home_ownership_digits,
  COUNTIF(REGEXP_CONTAINS(CAST(loan_intent AS STRING), r'[0-9]')) AS invalid_loan_intent_digits,
  COUNTIF(REGEXP_CONTAINS(CAST(loan_grade AS STRING), r'[0-9]')) AS invalid_loan_grade_digits,
  COUNTIF(SAFE_CAST(cb_person_default_on_file AS BOOL) IS NULL AND cb_person_default_on_file IS NOT NULL) AS invalid_default_on_file_format

FROM `sincere-cortex-467814-j3.bank_loan_credit_risk.credit_risk`;

# 5 -- Filter out extreme outliers and physiologically impossible records (e.g., age > 100, invalid employment length) and create the cleaned staging table
CREATE OR REPLACE TABLE `sincere-cortex-467814-j3.bank_loan_credit_risk.credit_risk_cleaned` AS
SELECT *
FROM `sincere-cortex-467814-j3.bank_loan_credit_risk.credit_risk`
WHERE person_age <= 100
  AND person_emp_length <= (person_age - 14)
  AND person_emp_length <= 60;

# 6 -- Generates the credit_risk_enriched staging table by adding calculated business logic: a binary default flag (is_default), income brackets, and loan burden categories based on standard financial risk benchmarks.
CREATE OR REPLACE TABLE `sincere-cortex-467814-j3.bank_loan_credit_risk.credit_risk_enriched` AS
SELECT 
    t.*,
    CASE WHEN t.loan_status = 1 THEN 1 ELSE 0 END AS is_default,
    CASE 
        WHEN t.person_income < 30000 THEN 'Low Income (<30k)'
        WHEN t.person_income BETWEEN 30000 AND 70000 THEN 'Medium Income (30k-70k)'
        ELSE 'High Income (>70k)'
    END AS income_bracket,
    CASE 
        WHEN t.loan_percent_income < 0.15 THEN 'Healthy (<15%)'
        WHEN t.loan_percent_income BETWEEN 0.15 AND 0.30 THEN 'Caution (15-30%)'
        ELSE 'Dangerous (>30%)'
    END AS burden_category
FROM `sincere-cortex-467814-j3.bank_loan_credit_risk.credit_risk_cleaned` t;
